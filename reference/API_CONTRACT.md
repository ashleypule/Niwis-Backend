# NIWIS Frontend ↔ Backend API Contract

> Generated by `scripts/generate-docs.ts` on 2026-10-09 — do not edit by hand.
> The response shapes are the zod schemas in [`src/types/api.ts`](../src/types/api.ts); the client validates every response
> against them (`src/api/client.ts`), and so does the mock API. A backend that conforms to this document can replace the
> mock by setting `VITE_DATA_SOURCE=api` and `VITE_API_BASE_URL`.

## Conventions

| Item | Rule |
| --- | --- |
| Base path | `/api/v1` (prefixed with `VITE_API_BASE_URL` when `VITE_DATA_SOURCE=api`) |
| Auth | `Authorization: Bearer <token>` when a token provider is registered (`setAuthTokenProvider` in `src/api/client.ts`) |
| Format | JSON, UTF-8. Numbers are SI / DWS units given in each payload (`unit`, `axes[].unit`, `columns[].unit`). Missing values are `null`, never `0`. |
| Time | ISO 8601. Monthly series are stamped on the first day of the month (`YYYY-MM-01`). Times are SAST (UTC+2). |
| Errors | Non-2xx with `{ "error": string, "issues"?: [] }`. 404 for an unknown module / metric / layer / entity. |
| Caching | Responses are cached client-side for 5 minutes (TanStack Query); send `Cache-Control` as appropriate. |

### Common query parameters (accepted by every module endpoint)

| Param | Values | Meaning |
| --- | --- | --- |
| `province` | `EC FS GP KZN LP MP NC NW WC` | Province scope |
| `wma` | `WMA1` … `WMA9` | Water Management Area scope (NWRS-3 nine WMAs) |
| `catchment` | quaternary / tertiary code, e.g. `C8` | Catchment scope |
| `municipality` | MDB code, e.g. `CPT`, `EC101` | Local / metro municipality scope |
| `period` | `latest 1M 3M 6M 12M 24M 5Y 10Y historical` | Look-back window (default `12M`) |
| `from`, `to` | `YYYY-MM-DD` | Custom range (overrides `period`) |
| `compare` | `1` | Include previous-period / last-year comparison series where available |
| `entityId` | entity id | Selected dam / station / borehole / plant for entity-specific datasets |

Scopes are hierarchical (`municipality` ⊂ `province`; `catchment` ⊂ `wma`); the most specific one wins. All aggregates must be computed from
the underlying entities (e.g. storage % = Σ volume / Σ capacity), never averaged from percentages.

## Endpoints

| Method & path | Response schema | Notes |
| --- | --- | --- |
| `GET /api/v1/{module}/summary` | `KpiSet` | Headline KPI cards. Each KPI carries `metric` (see Data Dictionary) so the UI knows units and which direction is good. |
| `GET /api/v1/{module}/timeseries?metric=` | `SeriesSet` | Time series (bands, dual axes, reference lines) |
| `GET /api/v1/{module}/forecast?metric=` | `SeriesSet` | Observed + median + p10–p90 band; `forecastStart` marks the divider |
| `GET /api/v1/{module}/breakdown?metric=` | `CategorySet` | Bars, stacked bars, donuts, waterfalls |
| `GET /api/v1/{module}/table?metric=` | `TableSet` | Tables; column `type`/`scale` drive formatting and class pills. `limit` optional. |
| `GET /api/v1/{module}/matrix?metric=` | `MatrixSet` | Heatmaps (rows × columns) |
| `GET /api/v1/{module}/map?layer=` | `MapLayer` | Choropleth (`geometry` + `values` by region id), grid or points |
| `GET /api/v1/{module}/entities` | `EntityList` | Selectable entities for the page selector (dams, stations …) |
| `GET /api/v1/{module}/entities/{id}` | `EntityDetail` | Info card for one entity |
| `GET /api/v1/alerts?module=&severity=&province=` | `AlertList` | Cross-module alerts / early warnings |
| `GET /api/v1/reference/{provinces\|wmas\|catchments\|municipalities}` | `RefList` | Cascading filter options |
| `GET /api/v1/search?q=` | `SearchResult` | Ctrl/⌘K search (places, dams, stations, settlements …) |

 * API response schemas (zod) and their TypeScript types. Every response from /api/v1 is validated against

import { z } from 'zod';

export const Num = z.number().nullable();

export const ScopeSchema = z.object({
  level: z.enum(['national', 'province', 'wma', 'catchment', 'municipality', 'entity']),
  id: z.string(),
  name: z.string(),
});
export type Scope = z.infer<typeof ScopeSchema>;

// ------------------------------------------------------------------ KPIs
export const KpiSchema = z.object({
  id: z.string(),
  /** metric registry id (config/metrics.ts) — drives units, decimals and delta colouring */
  metric: z.string(),
  label: z.string(),
  sublabel: z.string().optional(),
  value: Num,
  unit: z.string().optional(),
  decimals: z.number().optional(),
  /** categorical KPI: class id rendered as a status pill on `scale` */
  status: z.object({ scale: z.string(), id: z.string(), label: z.string() }).optional(),
  /** pre-formatted text shown instead of the number (e.g. "D2", "Moderate") */
  text: z.string().optional(),
  delta: Num.optional(),
  deltaUnit: z.string().optional(),
  deltaLabel: z.string().optional(),
  /** overrides the metric registry's goodDirection */
  goodDirection: z.enum(['up', 'down', 'neutral']).optional(),
  spark: z.array(z.number()).optional(),
});
export type Kpi = z.infer<typeof KpiSchema>;

export const KpiSetSchema = z.object({
  module: z.string(),
  asOf: z.string(),
  scope: ScopeSchema,
  kpis: z.array(KpiSchema),
});
export type KpiSet = z.infer<typeof KpiSetSchema>;

// ------------------------------------------------------------------ Time series / XY
const XVal = z.union([z.string(), z.number()]);
export const SeriesSchema = z.object({
  key: z.string(),
  name: z.string(),
  kind: z.enum(['line', 'bar', 'area', 'band', 'scatter', 'step']),
  axis: z.number().int().min(0).max(1).optional(),
  color: z.string().optional(),
  dash: z.boolean().optional(),
  stack: z.string().optional(),
  smooth: z.boolean().optional(),
  symbol: z.boolean().optional(),
  /** [x, y] or, for kind 'band', [x, low, high] */
  data: z.array(z.array(z.union([XVal, z.null()]))),
});
export type Series = z.infer<typeof SeriesSchema>;

export const AxisSchema = z.object({
  unit: z.string(),
  label: z.string().optional(),
  min: z.number().optional(),
  max: z.number().optional(),
  inverse: z.boolean().optional(),
  log: z.boolean().optional(),
});
export type Axis = z.infer<typeof AxisSchema>;

export const SeriesSetSchema = z.object({
  id: z.string(),
  title: z.string().optional(),
  xType: z.enum(['time', 'category', 'value']),
  xUnit: z.string().optional(),
  xLog: z.boolean().optional(),
  axes: z.array(AxisSchema).min(1).max(2),
  series: z.array(SeriesSchema),
  /** horizontal background bands, e.g. drought classes behind an SPI chart */
  bands: z.array(z.object({ from: z.number(), to: z.number(), color: z.string(), label: z.string() })).optional(),
  refLines: z
    .array(z.object({ y: z.number(), label: z.string(), axis: z.number().optional(), color: z.string().optional() }))
    .optional(),
  /** "Historical | Forecast" divider */
  asOf: z.string().optional(),
  note: z.string().optional(),
});
export type SeriesSet = z.infer<typeof SeriesSetSchema>;

// ------------------------------------------------------------------ Categories (bars, donuts, stacks, waterfalls)
export const CategorySetSchema = z.object({
  id: z.string(),
  title: z.string().optional(),
  unit: z.string(),
  categories: z.array(z.string()),
  /** optional per-category colour (donuts, class bars) */
  colors: z.array(z.string()).optional(),
  series: z.array(
    z.object({
      key: z.string(),
      name: z.string(),
      values: z.array(Num),
      color: z.string().optional(),
      stack: z.string().optional(),
      kind: z.enum(['bar', 'line']).optional(),
      axis: z.number().optional(),
    }),
  ),
  /** headline shown inside donuts */
  center: z.object({ value: z.string(), label: z.string() }).optional(),
  axis2Unit: z.string().optional(),
});
export type CategorySet = z.infer<typeof CategorySetSchema>;

// ------------------------------------------------------------------ Tables
export const ColumnSchema = z.object({
  key: z.string(),
  label: z.string(),
  type: z.enum(['text', 'number', 'class', 'trend', 'delta', 'bar', 'date', 'rank']),
  unit: z.string().optional(),
  decimals: z.number().optional(),
  /** palette id for 'class' columns (and optional colouring of 'number' columns) */
  scale: z.string().optional(),
  goodDirection: z.enum(['up', 'down', 'neutral']).optional(),
  align: z.enum(['left', 'right', 'center']).optional(),
});
export type Column = z.infer<typeof ColumnSchema>;

export const TableSetSchema = z.object({
  id: z.string(),
  title: z.string().optional(),
  columns: z.array(ColumnSchema),
  rows: z.array(z.record(z.union([z.string(), z.number(), z.null()]))),

});
export type TableSet = z.infer<typeof TableSetSchema>;

// ------------------------------------------------------------------ Matrix (heatmap)

  rows: z.array(z.string()),

  unit: z.string(),

});
export type MatrixSet = z.infer<typeof MatrixSetSchema>;

// ------------------------------------------------------------------ Map layers
export const MapPointSchema = z.object({
  id: z.string(),
  name: z.string(),
  lat: z.number(),
  lon: z.number(),
  value: z.union([z.number(), z.string(), z.null()]),
  /** marker radius hint 0–1 (e.g. dam capacity) */
  size: z.number().optional(),
  kind: z.string().optional(),
  sub: z.string().optional(),
});
export type MapPoint = z.infer<typeof MapPointSchema>;

export const MapLayerSchema = z.object({
  id: z.string(),
  title: z.string(),
  kind: z.enum(['choropleth', 'grid', 'points']),
  level: z.enum(['province', 'wma', 'catchment', 'municipality', 'grid', 'point']),
  scale: z.string(),
  unit: z.string(),
  date: z.string(),
  values: z.record(z.union([z.number(), z.string(), z.null()])).optional(),
  cellSize: z.number().optional(),
  cells: z.array(z.tuple([z.number(), z.number(), z.union([z.number(), z.null()])])).optional(),
  points: z.array(MapPointSchema).optional(),
});
export type MapLayer = z.infer<typeof MapLayerSchema>;

// ------------------------------------------------------------------ Entities
export const EntitySchema = z.object({
  id: z.string(),
  name: z.string(),
  type: z.string(),
  province: z.string().optional(),
  wma: z.string().optional(),
  municipality: z.string().optional(),
  lat: z.number().optional(),
  lon: z.number().optional(),
  sub: z.string().optional(),
});
export type Entity = z.infer<typeof EntitySchema>;

  title: z.string(),
  status: z.object({ scale: z.string(), id: z.string(), label: z.string() }).optional(),

    }),

    .optional(),
});
export type EntityDetail = z.infer<typeof EntityDetailSchema>;

// ------------------------------------------------------------------ Alerts, search, reference
export const SeveritySchema = z.enum(['critical', 'high', 'moderate', 'low', 'info']);
export type Severity = z.infer<typeof SeveritySchema>;

  date: z.string(),
  module: z.string(),
  severity: SeveritySchema,

  path: z.string(),
  params: z.record(z.string()).optional(),
});
export type SearchHit = z.infer<typeof SearchHitSchema>;

export type SearchResult = z.infer<typeof SearchResultSchema>;

export const RefItemSchema = z.object({
  id: z.string(),
  name: z.string(),
  parent: z.string().optional(),
  province: z.string().optional(),
  wma: z.string().optional(),
});
export type RefItem = z.infer<typeof RefItemSchema>;
export const RefListSchema = z.object({ type: z.string(), items: z.array(RefItemSchema) });
export type RefList = z.infer<typeof RefListSchema>;
